# =============================================================================
# SDB Framework — Core dependencies
# =============================================================================
# These packages are REQUIRED. Without them the module fails to import or the
# default CLI pipeline (`--mode full`) cannot run end-to-end.
#
# Install:
#   pip install -r requirements.txt
#
# For optional features (gradient boosting, Optuna HPO, publication plots,
# the neural mixture-of-experts model), also install:
#   pip install -r requirements-optional.txt
#
# Tested on Python 3.10 / 3.11. On Windows, rasterio/GDAL are far easier to
# install via conda-forge — see environment.yml instead of this file.
# =============================================================================

# --- Numerical / data handling ----------------------------------------------
numpy>=1.24,<3.0           # array backend used throughout the framework
pandas>=1.5,<3.0            # DataFrame I/O, feature tables, all report/CSV/Excel output
scipy>=1.10                 # PchipInterpolator (depth-error curve fit in ErrorPropagator);
                             # scipy.stats.wilcoxon (ModelComplexityLadder significance test)

# --- Machine learning core (scikit-learn) -----------------------------------
# Powers: RandomForestSDB, ExtraTreesSDB, SVR_SDB, MLP_SDB, ClusterBasedRegression
# (KMeans), GMMSoftMixtureSDB (GaussianMixture), LearnedGatingStrategy
# (RandomForestRegressor), RobustScaler/StandardScaler/SimpleImputer in
# FeatureEngineer, all metrics (RMSE/MAE/R2), KFold/LeaveOneOut in
# SpatialValidator, and SDBExplainer.compute_permutation_importance.
scikit-learn>=1.2

# --- Geospatial raster I/O ---------------------------------------------------
# Powers: RasterSampler (field-point band sampling) and RasterInferenceEngine
# (tiled GeoTIFF prediction + uncertainty raster). Requires GDAL under the hood.
rasterio>=1.3

# --- Model persistence --------------------------------------------------------
# Powers: SDBOperationalPipeline's final `.joblib` bundle export, and
# `load_deployment_bundle()` / `--mode predict`. Required for deployment, not
# just an optional convenience.
joblib>=1.2

# --- Tabular file I/O ----------------------------------------------------------
# Powers: RasterSampler.read_field_data (.xlsx field data), the multi-sheet
# "article results" workbook writer, and workbook styling (openpyxl.styles /
# formatting.rule / utils used in _style_article_workbook).
openpyxl>=3.1

# --- Configuration files -------------------------------------------------------
# Powers: SDBConfig.save() / SDBConfig.from_yaml() and the `--config file.yaml`
# CLI option. Falls back to JSON if absent, but YAML config files will not
# load without it.
PyYAML>=6.0
